\subsection{Mathematical statement of the claimed result}
\paragraph{Result 1.} Let $M=\lfloor\alpha N\rfloor$ and let $A$ have independent entries of law
\[
 \nu_\varepsilon=\tfrac12N(0,1-\varepsilon)+\tfrac12N(0,1+\varepsilon).
\]
For $\|x\|^2=N$, set $g_\mu=A_\mu x/\sqrt N$, $h_\mu=1+g_\mu$, and $H_N(x)=\frac12\sum_{\mu=1}^M(-h_\mu)_+^2$. Use spherical Gibbs probability proportional to $e^{-\beta H_N}$. SAT means that $H_N$ vanishes somewhere on the sphere. For $0<\delta<u$ and $s>0$, define
\[
 G_{N,\beta,\alpha}(\delta,u)=\mathbb E\left\langle M^{-1}\sum_\mu\mathbf1_{\{\delta<h_\mu\le u\}}\right\rangle.
\]
On $C(x)=\{\mu:h_\mu<0\}$ put $\bar r=|C|^{-1}\sum_{\mu\in C}(-h_\mu)$ and $f_\mu=-h_\mu/\bar r$. Set
\[
 F_{N,\beta,\alpha}(s)=\mathbb E\left\langle |C|^{-1}\sum_{\mu\in C}\mathbf1_{\{f_\mu\le s\}}\right\rangle,
\]
with value zero when $C$ is empty. There exist $\alpha_c>0$ and $0<\varepsilon_0<1$ such that, for each fixed $0\le\varepsilon<\varepsilon_0$, SAT probability tends to one for $\alpha<\alpha_c$ and zero for $\alpha>\alpha_c$. The ordered limits
\[
 G_J(u)=\lim_{\delta\downarrow0}\lim_{\alpha\downarrow\alpha_c}\lim_{\beta\to\infty}\lim_{N\to\infty}G_{N,\beta,\alpha}(\delta,u),\quad
 F_J(s)=\lim_{\alpha\downarrow\alpha_c}\lim_{\beta\to\infty}\lim_{N\to\infty}F_{N,\beta,\alpha}(s)
\]
exist and are independent of $\varepsilon$. The second is a probability distribution function on $(0,\infty)$, of mean one and with no atom at zero. As $u,s\downarrow0$,
\[
 G_J(u)=u^{1-\gamma+o(1)},\quad F_J(s)=s^{1+\theta+o(1)},\quad
 \gamma=(2+\theta)^{-1},\quad
 0.4126930<\gamma<0.4126934,\quad 0.4231063<\theta<0.4231088.
\]
