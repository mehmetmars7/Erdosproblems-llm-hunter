\section*{The thermodynamic pressure of every bounded Gaussian Ising perceptron activation}

\subsection{OpenAI's claim}
OpenAI presents a proof in the selected manuscript. The Ising Gaussian perceptron is claimed to have an exact pressure formula for every bounded Borel log-potential at positive temperature and density.

\subsection{Mathematical statement of the claimed result}
\paragraph{Result 1.} Fix $\alpha>0$ and any bounded Borel $f:\mathbb R\to\mathbb R$. Let $M=\lfloor\alpha N\rfloor$, and take independent standard Gaussian vectors $g^1,\ldots,g^M\in\mathbb R^N$. With uniform spin probability,
\[
 p_N(f)=N^{-1}\log\left(2^{-N}\sum_{x\in\{-1,1\}^N}
 \exp\sum_{a=1}^Mf(g^a\cdot x/\sqrt N)\right).
\]
For $G\sim N(0,1)$ define
\[
 \mathcal T_{s,d}U(x)=\begin{cases}d^{-1}\log\mathbb Ee^{dU(x+\sqrt sG)},&d>0,\\
 \mathbb EU(x+\sqrt sG),&d=0.\end{cases}
\]
Let $\mathcal Q$ contain nondecreasing paths $q:(0,1)\to[0,1]$. For a step path with values $q_i$ on $\zeta_i<u<\zeta_{i+1}$, $0=\zeta_0<\cdots<\zeta_k=1$, append $q_{-1}=0,q_k=1$, start $U_k=f$, and recurse $U_{i-1}=\mathcal T_{q_i-q_{i-1},\zeta_i}U_i$, $i=k,\ldots,0$. Define $V_f(q)=U_{-1}(0)$, using its continuous path extension for general $q$. For a finite nonnegative nondecreasing step field $h$, instead start $U_{k-1}(x)=\log\cosh x-h_{k-1}/2$, append $h_{-1}=0$, and recurse $U_{i-1}=\mathcal T_{h_i-h_{i-1},\zeta_i}U_i$. Put $\ell(h)=U_{-1}(0)$ and
\[
 S_{\mathrm I}(q)=\sup_h\left\{\ell(h)+\frac12\int_0^1h(u)q(u)\,du\right\}.
\]
The finite limiting value is
\[
 \lim_N\mathbb Ep_N(f)=\inf_{q\in\mathcal Q}\{\alpha V_f(q)+S_{\mathrm I}(q)\}=:\mathcal P_f(\alpha),
 \qquad p_N(f)\xrightarrow{\mathbb P}\mathcal P_f(\alpha).
\]
In particular $f=\beta\phi$ is allowed for every $\beta>0$ and bounded continuous $\phi$.
