\[
 \psi(x,y)+\psi(y,x)=0,\qquad
 \psi(x,y)+\psi(y,-x-y)+\psi(-x-y,x)=0,
\]
and the pentagon
\[
 \psi(t_{12},t_{23}+t_{24})+\psi(t_{13}+t_{23},t_{34})
 =\psi(t_{23},t_{34})+\psi(t_{12}+t_{13},t_{24}+t_{34})+\psi(t_{12},t_{23})
\]
in the infinitesimal pure braid Lie algebra, defined by $t_{ij}=t_{ji}$, $[t_{ij},t_{k\ell}]=0$ on distinct indices and $[t_{ij},t_{ik}+t_{jk}]=0$. Define $D_{\psi}(x)=0$, $D_{\psi}(y)=[y,\psi]$ and Ihara bracket $\{\psi,\phi\}=D_{\psi}\phi-D_{\phi}\psi+[\psi,\phi]$. The claim gives homogeneous $\sigma_{2k+1}\in W_{2k+1}$ for every $k\ge1$ with
\[
 \operatorname{Lie}_{\mathbb Q}\langle e_3,e_5,e_7,\ldots\rangle
 \xrightarrow{\sim}(W,\{\ ,\ \}),\qquad e_{2k+1}\mapsto\sigma_{2k+1}.
\]
It asserts generation as well as freeness, bracket closure and an induced continuous isomorphism of weight completions.
