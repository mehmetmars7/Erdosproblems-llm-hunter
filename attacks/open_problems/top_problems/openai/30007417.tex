\subsection{Mathematical statement of the claimed result}
\paragraph{Result 1.} For angles $\theta_x\in\mathbb R/(2\pi\mathbb Z)$ on $\Lambda_R=[-R,R]^2\cap\mathbb Z^2$, use the free-boundary cosine law proportional to $\exp[b\sum_{\{x,y\}\subset\Lambda_R,\,|x-y|=1}\cos(\theta_x-\theta_y)]$, relative to product uniform angles. Count each edge once and omit exterior edges. Define
\[
 C_b(r)=\lim_{R\to\infty}\mathbb E_{R,b}^{\mathrm{free}}\cos(\theta_0-\theta_{re_1}),\quad
 m(b)=\lim_{r\to\infty}-r^{-1}\log C_b(r),\quad b_c=\inf\{b>0:m(b)=0\}.
\]
The thermodynamic limit precedes the separation limit. At the critical point,
\[
 \lim_{n\to\infty}\frac{\log C_{b_c}(n)}{\log n}=-\frac14.
\]
Equivalently, for every $\varepsilon>0$ and all sufficiently large integers $n$, $n^{-1/4-\varepsilon}\le C_{b_c}(n)\le n^{-1/4+\varepsilon}$. This paper determines the power exponent; a separate selected manuscript supplies the logarithmic factor.

\paragraph{Result 2.} For angles $\theta_x\in\mathbb R/(2\pi\mathbb Z)$ on $\Lambda_R=[-R,R]^2\cap\mathbb Z^2$, use the free-boundary cosine law proportional to $\exp[b\sum_{\{x,y\}\subset\Lambda_R,\,|x-y|=1}\cos(\theta_x-\theta_y)]$, relative to product uniform angles. Count each edge once and omit exterior edges. Define
\[
 C_b(r)=\lim_{R\to\infty}\mathbb E_{R,b}^{\mathrm{free}}\cos(\theta_0-\theta_{re_1}),\quad
 m(b)=\lim_{r\to\infty}-r^{-1}\log C_b(r),\quad b_c=\inf\{b>0:m(b)=0\}.
\]
The thermodynamic limit precedes the separation limit. There exists a finite positive $B_{\mathrm{XY}}$ satisfying
\[
 C_{b_c}(r)=B_{\mathrm{XY}}r^{-1/4}(\log r)^{1/8}(1+o(1))\quad(r\to\infty\text{ through positive integers}).
\]
This fixes the logarithmic multiplier as well as the critical power.
