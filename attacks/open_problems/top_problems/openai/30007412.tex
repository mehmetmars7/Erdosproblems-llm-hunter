\subsection{Mathematical statement of the claimed result}
\paragraph{Result 1.} On square-lattice bonds let $J_e=1+\varepsilon\xi_e$, with independent fair signs $\xi_e$. Define the physical critical inverse temperature by
\[
 \beta_c(\varepsilon)=\inf\left\{\beta\ge0:
 \mathbb E_{\mathrm{env}}\lim_{m\to\infty}\langle\sigma_0\rangle^+_{[-m,m]^2,\beta,J}>0\right\},
\]
normalizing each Gibbs law before environmental averaging. Let $C_{\varepsilon,\omega}(r)=\lim_{n\to\infty}\langle\sigma_0\sigma_{(r,0)}\rangle^{\mathrm{free}}_{[-n,n]^2,\beta_c(\varepsilon),\omega}$ and
\[
 R_\varepsilon(r)=\frac{\mathbb E C_{\varepsilon,\omega}(r)^2}{(\mathbb E C_{\varepsilon,\omega}(r))^2}.
\]
The theorem takes deterministic critical-reference correlation, conditional-averaging and marked-source estimates as inputs. Its conditional-averaging input can be expressed with $P_RF=\mathbb E[F\mid\text{cutting spins of }[-R,R]^2]$, $F^\circ=F-\langle F\rangle_0$ and $d_R(F)=\langle F\rangle^+_{\mathrm{disk}(4R)}-\langle F\rangle_0$:
\[
 |d_R(F)|\le C(w/R)^b\|F\|_\infty,quad
 \|P_RF^\circ\|_\infty\le C|d_R(F)|+C(w/R)^p\|F\|_\infty,
\]
for $F$ supported in a radius-$w$ square, sufficiently large fixed $R/w$, and fine scales. The pairs $(b,p)$ are $(1,3)$ for even $F$ invariant under right-angle rotations, $(1,2)$ for other even $F$, and $(1/8,9/8)$ for odd $F$. The energy and spin calibrations have nonzero limits $R^bd_R(\mathcal O_b)$, where $\mathcal O_1$ is the average of four centred incident bond products and $\mathcal O_{1/8}=\sigma_0$. Subject to these deterministic inputs and their bulk field-product estimates, some $\varepsilon_0>0$ gives, for fixed $0<\varepsilon<\varepsilon_0$,
\[
 \lim_{r\to\infty}\frac{\log R_\varepsilon(r)}{\log\log r}=\frac14.
\]
The physical critical temperature is the unique positive root of
\[
 \sinh(2\beta(1-\varepsilon))\sinh(2\beta(1+\varepsilon))=1.
\] The deterministic calibration input also requires, for $b\in\{1,1/8\}$,
\[
 c\le\|w^bP_w\mathcal O_b\|_\infty\le C,\qquad
 c(w/R)^b\le|d_R(w^bP_w\mathcal O_b)|\le C(w/R)^b,
\]
with the same bulk scale and buffer conventions as the operator estimates.
