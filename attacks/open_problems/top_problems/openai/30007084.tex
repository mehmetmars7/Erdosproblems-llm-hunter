\section*{Colin de Verdière chromatic-number conjecture}

\subsection{OpenAI's claim}
\textit{Colin de Verdière chromatic obstruction}: The manuscript claims finite graphs whose fractional chromatic number is greater than $\mu(G)+1$, and therefore whose ordinary chromatic number violates the same bound. Such examples exist at arbitrarily large orders.

\subsection{Mathematical statement of the claimed result}
\paragraph{Mathematical claim 1.} The claimed counterexamples are finite simple graphs of arbitrarily large order $m$ with \[\alpha(G)\le2,\qquad\mu(G)+1<m/2\le\chi(G),\] where $\mu(G)$ is the Colin de Verdi\`ere spectral graph invariant. The fractional-coloring corollary strengthens the obstruction to \[\chi_f(G)\ge m/\alpha(G)\ge m/2>\kappa(G)+1\ge\mu(G)+1\ge h(G).\] Here $\kappa$ is its strong-Arnold-property-free spectral relaxation, and $h$ is the largest complete-minor order. In particular both $\chi\le\mu+1$ and $\chi_f\le h$ are refuted. A further corollary gives $\chi_{\mathrm{list}}(G)\le C(\mu(G)+1)$ with an absolute $C$.
