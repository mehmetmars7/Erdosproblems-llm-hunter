\subsection{OpenAI's claim}

\subsection{Mathematical statement of the claimed result}
\paragraph{Result 1.} For $N=2^d$ cards, one physical Thorp shuffle pairs corresponding cards in the two deck halves, independently orders each pair by a fair coin, and interleaves. Its law is $q_d$ on $S_N$; $U_N$ is uniform. A sweep uses each binary-coordinate matching once with independent fair switches, hence costs $d$ physical shuffles. Total variation means half the sum of absolute mass differences. If $T_N(\lambda)=\mathbb E\rho_\lambda(g)$ is the sweep average on the irreducible unitary $S_N$-representation of degree $D_\lambda$, put $B_N(\lambda)=T_N(\lambda)T_N(\lambda)^*$. Absolute $\eta>0$ and an even integer $r\ge2$ satisfy
\[
 D_\lambda\operatorname{Tr}B_N(\lambda)^r\le D_\lambda^{1-\eta}
 \quad(\lambda\vdash N),
\]
with unnormalized trace. Consequently a fixed number $q$ of independent forward sweeps has distance from $U_N$ tending to zero as $N\to\infty$, uniformly over initial decks.

\paragraph{Result 2.} Let $B$ be convolution by a probability measure on a finite group $G$, and $B_\rho$ its Fourier matrix in an irreducible unitary representation of dimension $D_\rho$. Suppose fixed $p>0$, $0<\varepsilon\le1/2$ satisfy
\[
 \sum_{\rho\ne\mathbf1}D_\rho\operatorname{Tr}|B_\rho|^p\le\varepsilon,
 \qquad |B_\rho|=(B_\rho^*B_\rho)^{1/2}.
\]
For a sequence of such groups, assume every nonzero nonconstant block has dimension at least $D_{\min}\to\infty$. If $\mu_M$ is the law after $M$ independent convolution steps and $2M>p$, then
\[
 4\|\mu_M-U_G\|_{\mathrm{TV}}^2
 \le\varepsilon(\varepsilon/D_{\min})^{(2M-p)/p}\longrightarrow0.
\]
This is a conditional spectral-conversion result; its moment hypothesis must be supplied by the sweep estimates.
