\subsection{Mathematical statement of the claimed result}
\paragraph{Manuscript 1.}
For integers $n,m\ge1$, $d\ge0$, a complex matrix $A\in M_n(\mathbb C)$ and coefficients $B_k\in M_m(\mathbb C)$, let
\[
 P(z)=\sum_{k=0}^d B_kz^k,\qquad P[A]=\sum_{k=0}^d A^k\otimes B_k,
 \quad W(A)=\{x^*Ax:x\in\mathbb C^n,\ \|x\|_2=1\}.
\]
With Hilbert-space operator norms, the claimed complete inequality is
\[
 \|P[A]\|\le2\max_{z\in W(A)}\|P(z)\|.
\]
The factor $2$ is asserted sharp uniformly over all three integer parameters. Matrix-valued coefficients are allowed; the scalar polynomial case follows with $m=1$.

\paragraph{Manuscript 2.}
For $n,m\ge1$, $A\in M_n(\mathbb C)$, $d\ge0$ and arbitrary $B_0,\ldots,B_d\in M_m(\mathbb C)$, write $P(z)=\sum_jB_jz^j$ and $W(A)=\{x^*Ax:\|x\|_2=1\}$. The claimed complete polynomial inequality is
\[
 \left\|\sum_{j=0}^dA^j\otimes B_j\right\|\le2\max_{z\in W(A)}\|P(z)\|.
\]
Hilbert-space operator norms are used, and the constant two is claimed sharp independently of matrix sizes and polynomial degree. The scalar case takes $m=1$.
