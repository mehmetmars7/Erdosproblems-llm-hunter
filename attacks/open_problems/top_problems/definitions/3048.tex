\section{Sequence defined on multisets}\label{3048}
{\small\sffamily UnsolvedMath ID 3048 · Combinatorics}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For $m\geq1$ let $A=\begin{pmatrix}a_1&\cdots&a_m\\b_1&\cdots&b_m\end{pmatrix}$, where $1\leq a_1<\cdots<a_m$ and $b_j\in\mathbb N$. Let $c_1<\cdots<c_q$ list the distinct integers appearing in either row. Define $T(A)$ to have first row $(c_1,\ldots,c_q)$ and second row $(\mu_A(c_1),\ldots,\mu_A(c_q))$, where $\mu_A(c)=|\{j:a_j=c\}|+|\{j:b_j=c\}|$. Set $A_0=A$ and $A_{r+1}=T(A_r)$ for $r\geq0$.
\paragraph{Statement recorded in the supplied source.}
Does every initial array admit integers $r_0\geq0$ and $p\in\{1,2,3\}$ such that $A_{r+p}=A_r$ for every $r\geq r_0$? \sref{3048}{2}
\subsection{Short English statement}
Does repeatedly replacing the entries of a two-row array by the list of distinct entries and their frequencies always eventually cycle with period at most three?
