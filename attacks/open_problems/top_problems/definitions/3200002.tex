\section{Dittert–Hajek conjecture}\label{3200002}
{\small\sffamily UnsolvedMath ID 3200002 · Combinatorics}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For $n\ge2$, put
\[
\mathcal K_n=\left\{A=(a_{ij})\in[0,\infty)^{n\times n}:\ \sum_{i,j=1}^{n}a_{ij}=n\right\}.
\]
For $A\in\mathcal K_n$ define row sums $r_i=\sum_j a_{ij}$, column sums $c_j=\sum_i a_{ij}$, and
\[
\operatorname{per}(A)=\sum_{\sigma\in S_n}\prod_{i=1}^{n}a_{i,\sigma(i)},\qquad
\phi(A)=\prod_{i=1}^{n}r_i+\prod_{j=1}^{n}c_j-\operatorname{per}(A),
\]
where $S_n$ is the set of permutations of $\{1,\ldots,n\}$. Let $J_n$ be the all-ones matrix and $U_n=J_n/n$. Thus $\phi(U_n)=2-n!/n^n$. The case $n=1$ has just one admissible matrix.
\paragraph{Statement recorded in the supplied source.}
Is it true for every $n\ge2$ and $A\in\mathcal K_n$ that
\[
\phi(A)\le 2-\frac{n!}{n^n},\qquad
\phi(A)=2-\frac{n!}{n^n}\ \Longleftrightarrow\ A=U_n?
\] \sref{3200002}{2}
\subsection{Short English statement}
Among nonnegative square matrices whose entries sum to the size of the matrix, prove that the Dittert functional has its unique maximum at the constant matrix.
