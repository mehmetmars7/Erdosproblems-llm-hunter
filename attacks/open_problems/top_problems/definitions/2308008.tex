\section{Research Problems in Function Theory — Problem 8.8}\label{2308008}
{\small\sffamily UnsolvedMath ID 2308008 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. For $p>0$, the Bergman space is $A^p(\mathbb D)=\{f\text{ analytic}:\|f\|_{A^p}=(\int_{\mathbb D}|f|^p\,dA)^{1/p}<\infty\}$, where $dA=dx\,dy$ is ordinary area measure. For a relatively closed $F\subset\mathbb D$, define $\|f\|_F=\sup_{z\in F}|f(z)|\in[0,\infty]$, with $\|f\|_\varnothing=0$, and its $A^2$ hull \[\widehat F_{A^2}=\{z\in\mathbb D:\ |f(z)|\le\|f\|_F\text{ for every }f\in A^2(\mathbb D)\}.\]
\paragraph{Statement recorded in the supplied source.}
Give a geometric description of $\widehat F_{A^2}$ for every relatively closed $F\subset\mathbb D$. \sref{2308008}{2}
\subsection{Short English statement}
Describe geometrically the hull of a relatively closed disk set defined by all Bergman functions.
