\section{Research Problems in Function Theory — Problem 5.10}\label{2305010}
{\small\sffamily UnsolvedMath ID 2305010 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. All Taylor coefficients below are taken at $0$. Let $S$ be the class of holomorphic injective functions $g:\mathbb D\to\mathbb C$ with $g(0)=0$ and $g'(0)=1$.Let $S_{\mathrm{star}}\subset S$ consist of those functions for which $g(\mathbb D)$ is starlike about $0$: $tw\in g(\mathbb D)$ whenever $w\in g(\mathbb D)$ and $0\le t\le1$. Put
\[\ell(g)=\sup\{\rho>0:B(w,\rho)\subset g(\mathbb D)\text{ for some }w\in\mathbb C\},\qquad
B_{\mathrm{sch}}=\inf_{g\in S}\ell(g),\quad B_{\mathrm{star}}=\inf_{g\in S_{\mathrm{star}}}\ell(g).\]
The source denotes these two constants by $S$ and $S_1$, respectively.
\paragraph{Statement recorded in the supplied source.}
Determine both $B_{\mathrm{sch}}$ and $B_{\mathrm{star}}$. \sref{2305010}{2}
\subsection{Short English statement}
Find the exact universal image-disc radius for normalized univalent functions, and also for the subclass with starlike images.
