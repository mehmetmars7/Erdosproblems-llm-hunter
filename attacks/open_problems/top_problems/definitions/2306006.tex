\section{Research Problems in Function Theory — Problem 6.6}\label{2306006}
{\small\sffamily UnsolvedMath ID 2306006 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $\mathbb D=\{z\in\mathbb C:|z|<1\}$, $\mathbb T=\partial\mathbb D$, and $\mathbb D^*=\{z\in\mathbb C:|z|>1\}$. Let $\Sigma$ be the class of analytic injective functions on $\mathbb D^*$ with Laurent expansion $F(z)=z+\sum_{n=1}^{\infty}b_nz^{-n}$. The inverse branch of $F$ fixing infinity has a Laurent expansion near infinity \[G(w)=F^{-1}(w)=w+\sum_{n=1}^{\infty}c_nw^{-n}.\] Its expansion need not be defined on all of $\mathbb D^*$. The source records $|c_n|\le2^n/n$, and that for every $\varepsilon>0$ the bound \[|c_{2n-1}|>(1-\varepsilon)\frac{2^{2n-2}e}{(\pi n^3)^{1/2}}\] is possible for all sufficiently large $n$.
\paragraph{Statement recorded in the supplied source.}
Determine the sharp orders of magnitude of the inverse coefficients $c_n$, including matching upper bounds and attainable lower bounds as $n\to\infty$. \sref{2306006}{2}
\subsection{Short English statement}
Determine the sharp growth rates of coefficients of inverse branches of exterior normalized univalent functions.
