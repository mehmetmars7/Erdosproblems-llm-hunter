\section{Research Problems in Function Theory — Problem 5.23}\label{2305023}
{\small\sffamily UnsolvedMath ID 2305023 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. All Taylor coefficients below are taken at $0$. Let $S$ be the class of holomorphic injective functions $g:\mathbb D\to\mathbb C$ with $g(0)=0$ and $g'(0)=1$.Write $S_0$ for the unnormalized class of univalent functions with $f(0)=0$, and $C\subset S$ for functions with convex images. Let $K\subset S$ be the close-to-convex class: $f\in K$ if for some $g\in C$ and $\theta\in\mathbb R$, $\operatorname{Re}(e^{i\theta}f'(z)/g'(z))>0$ throughout $\mathbb D$. The Nevanlinna class $N$ consists of holomorphic $f$ with
\[\sup_{0<r<1}\frac1{2\pi}\int_0^{2\pi}\log^+|f(re^{it})|\,dt<\infty.
\]
Its Smirnov subclass $N^+$ consists of those $f\in N$ for which these boundary functions $\log^+|f(re^{it})|$ are uniformly integrable as $r\uparrow1$: for every $\varepsilon>0$ there is $\eta>0$ such that their integrals over any measurable set of length less than $\eta$ are less than $\varepsilon$, uniformly for $r$ sufficiently near $1$. Here $\log^+x=\max(0,\log x)$ and $\log^+0=0$. A coefficient multiplier from a class $A$ to a class $B$ is a sequence $(\lambda_n)_{n\ge0}$ for which $T_\lambda f(z)=\sum_{n\ge0}\lambda_n a_nz^n$ converges on $\mathbb D$ and belongs to $B$ whenever $f(z)=\sum_{n\ge0}a_nz^n\in A$.
\paragraph{Statement recorded in the supplied source.}
Characterize all coefficient multiplier sequences for each of the following mappings:
\[S\to S,\qquad S_0\to S_0,\qquad K\to K,\qquad S\to C,\qquad N\to N,\qquad N^+\to N^+.
\]
For the normalized classes, preservation includes the normalization at $0$; for $S_0$ it only requires a nonzero first coefficient. \sref{2305023}{2}
\subsection{Short English statement}
Describe coefficient multipliers preserving normalized or unnormalized univalence, close-to-convexity, bounded characteristic and the Smirnov class, and those sending univalent functions to convex ones.
