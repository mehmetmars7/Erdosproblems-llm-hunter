\section{Research Problems in Function Theory — Problem 8.18}\label{2308018}
{\small\sffamily UnsolvedMath ID 2308018 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $D\subset\mathbb C$ be a simply connected domain containing $0$. For analytic $f$ on $D$ with $f(0)=0$, define \[E_D(f)=\int_D|f^{\prime}|^2\,dA,\qquad N_D(f)=\int_D|f|^2\,dA,\] permitting the value $+\infty$. The analytic Poincar\'e inequality means that there is $K_D<\infty$ such that $N_D(f)\le K_DE_D(f)$ whenever $E_D(f)<\infty$; functions with infinite energy give no restriction.
\paragraph{Statement recorded in the supplied source.}
Give necessary and sufficient geometric conditions on $D$ for this inequality to hold for every such analytic $f$. \sref{2308018}{2}
\subsection{Short English statement}
Characterize the simply connected domains that satisfy the analytic Poincaré inequality for functions vanishing at the origin.
