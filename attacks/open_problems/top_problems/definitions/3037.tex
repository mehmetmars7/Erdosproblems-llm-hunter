\section{Inequality for square summable complex series}\label{3037}
{\small\sffamily UnsolvedMath ID 3037 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $\ell^2(\mathbb C)=\{\alpha=(\alpha_n)_{n\geq1}:\sum_{n\geq1}|\alpha_n|^2<\infty\}$. For $k\geq0$ define
\[T_k(\alpha)=\sum_{\ell\geq0}\frac{\alpha_{2^k(2\ell+1)}}{\ell+1}.\]
Each inner series converges absolutely by Cauchy--Schwarz, since $\sum_{\ell\geq0}(\ell+1)^{-2}=\pi^2/6$ and its selected coefficients are square summable.
\paragraph{Statement recorded in the supplied source.}
For every $\alpha\in\ell^2(\mathbb C)$, is
\[\sum_{n\geq1}|\alpha_n|^2\geq\frac{6}{\pi^2}\sum_{k\geq0}\left|\sum_{\ell\geq0}\frac{\alpha_{2^k(2\ell+1)}}{\ell+1}\right|^2\]
valid? \sref{3037}{2}
\subsection{Short English statement}
Does the stated weighted sum along the odd-part dyadic partition define a squared-norm bound with constant six over pi squared?
