\section{Some Open Problems in Elasticity — Existence of minimizers}\label{2000001}
{\small\sffamily UnsolvedMath ID 2000001 · Partial Differential Equations}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Let $\Omega\subset\mathbb R^3$ be a bounded Lipschitz domain. Write $\partial\Omega=\Gamma_D\cup\Gamma_N\cup N$, where $\Gamma_D,\Gamma_N$ are disjoint relatively open subsets, $\mathcal H^2(N)=0$, and $\mathcal H^2(\Gamma_D)>0$. Here $\mathcal H^2$ is two-dimensional Hausdorff measure. A deformation $y:\Omega\to\mathbb R^3$ belongs to the Sobolev space $W^{1,1}(\Omega;\mathbb R^3)$, so its weak derivative $Dy$ is defined almost everywhere. Prescribe $y=\bar y$ on $\Gamma_D$ in the trace sense and zero applied traction on $\Gamma_N$.
Let $\mathbb M_+^{3\times3}=\{A\in\mathbb R^{3\times3}:\det A>0\}$. The homogeneous stored-energy density $W:\mathbb M_+^{3\times3}\to[0,\infty)$ is $C^1$, satisfies frame indifference $W(RA)=W(A)$ for $R\in\operatorname{SO}(3)$, and satisfies
\[
 W(A)\longrightarrow+\infty\quad\text{as}\quad\det A\longrightarrow0^+.
\]
Extend $W$ by $+\infty$ when $\det A\le0$. Put
\[
 I(y)=\int_\Omega W(Dy(x))\,dx,\qquad
 \mathcal A=\{y\in W^{1,1}(\Omega;\mathbb R^3):I(y)<\infty,\ y|_{\Gamma_D}=\bar y\}.
\]
Assume $\mathcal A\ne\varnothing$. For a matrix $A$, $\operatorname{cof}A$ denotes its cofactor matrix and $|A|$ its Frobenius norm.
The extended density is \emph{quasiconvex} when, for every matrix $A$, bounded open set $U\subset\mathbb R^3$ and $\varphi\in C_c^\infty(U;\mathbb R^3)$,
\[
 |U|W(A)\le\int_U W(A+D\varphi)\,dx.
\]
The inequality is interpreted for extended real values. Assume also the source's coercivity hypothesis: for constants $c_0>0$ and $c_1\ge0$,
\[
 W(A)\ge c_0\bigl(|A|^2+|\operatorname{cof}A|^{3/2}\bigr)-c_1.
\]
\paragraph{Statement recorded in the supplied source.}
Does $I$ attain its infimum on $\mathcal A$ for every quasiconvex density and admissible boundary data satisfying these hypotheses? More generally, establish quasiconvex existence theorems compatible with the determinant blow-up condition, rather than upper polynomial growth on all matrices. \sref{2000001}{2}
\subsection{Short English statement}
Prove existence of an energy-minimizing deformation when quasiconvexity replaces polyconvexity, while keeping the physically necessary infinite energy as the determinant tends to zero.
