\section{Research Problems in Function Theory — Problem 5.42}\label{2305042}
{\small\sffamily UnsolvedMath ID 2305042 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. All Taylor coefficients below are taken at $0$.Let $W^+$ be the analytic Wiener algebra: $f(z)=\sum_{n\ge0}a_nz^n$ with $\|f\|_{W^+}=\sum_{n\ge0}|a_n|<\infty$. For $0<\delta\le1$, define
\[V(\delta)=\sup\left\{\sum_{n\ge0}|b_n|:\|f\|_{W^+}=1,\ |f(z)|\ge\delta\ (z\in\mathbb D),\ \frac1{f(z)}=\sum_{n\ge0}b_nz^n\right\},
\]
allowing the supremum to be $\infty$. Set $\delta_* =\inf\{\delta\in(0,1]:V(\delta)<\infty\}$. The bound in $V(\delta)$ is uniform over all admissible $f$ and may depend only on $\delta$.
\paragraph{Statement recorded in the supplied source.}
Determine $\delta_*$. In particular, is $\delta_*=1/2$? \sref{2305042}{2}
\subsection{Short English statement}
Determine the smallest lower-modulus threshold above which normalized absolutely convergent power series have a uniform absolute-coefficient bound for their reciprocals.
