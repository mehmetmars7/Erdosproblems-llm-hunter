\section{Research Problems in Function Theory — Problem 6.48}\label{2306048}
{\small\sffamily UnsolvedMath ID 2306048 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. Let $S$ consist of holomorphic injective functions $f(z)=z+\sum_{n\ge2}a_nz^n$ on $\mathbb D$; set $a_0=0$, $a_1=1$. A function in $S$ is starlike if its image is starlike about $0$, and convex if its image is a convex domain. For $0<\alpha<1$, let $S^*_{\mathrm{str}}(\alpha)=\{f\in S:|\arg(zf'(z)/f(z))|<\alpha\pi/2\text{ on }\mathbb D\}$; extend the quotient by $1$ at $0$. The argument is the continuous branch equal to $0$ at $0$. For $\alpha>0$, the normalized Bazilevi\v{c} class $B(\alpha)$ comprises $f\in S$ for which there exist a normalized starlike $s\in S$ and a holomorphic $p$ with $p(0)=1$, $\operatorname{Re}p>0$, satisfying
\[\frac{zf'(z)(f(z)/z)^{\alpha-1}}{z(s(z)/z)^\alpha}=p(z).\]
The powers use logarithms normalized to vanish at $0$, and the expression extends to $0$ by $1$. Equivalently $f(z)=z\left[\alpha\int_0^1p(tz)(s(tz)/(tz))^\alpha t^{\alpha-1}\,dt\right]^{1/\alpha}$, with its normalized branch; the factor $\alpha$ enforces $f'(0)=1$. For $0<\alpha<1$, define the Mocanu--Reade class $M(\alpha)\subset S$ by
\[\operatorname{Re}\left[\alpha\left(1+\frac{zf''}{f'}\right)+(1-\alpha)\frac{zf'}{f}\right]>0\quad(z\in\mathbb D).\]
\paragraph{Statement recorded in the supplied source.}
For each indicated parameter, determine the sharp bounds $\sup_{f\in A}|a_n|$ for all $n\ge2$, separately for $A=B(\alpha)$, $A=M(\alpha)$ and $A=S^*_{\mathrm{str}}(\alpha)$. \sref{2306048}{2}
\subsection{Short English statement}
Find sharp coefficient bounds for the normalized Bazilevič, Mocanu–Reade and strongly starlike one-parameter families.
