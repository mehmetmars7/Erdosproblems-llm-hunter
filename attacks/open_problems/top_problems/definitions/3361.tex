\section{Alexa's Conjecture on Primality}\label{3361}
{\small\sffamily UnsolvedMath ID 3361 · Number Theory}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For each integer $p\ge8$, put $k(p)=\lfloor p^{1/3}/2\rfloor$. For $1\le i\le k(p)$, let $r_i(p)$ be the least nonnegative residue of $(2i+1)^{p-1}$ modulo $p$, so that $0\le r_i(p)<p$ and $r_i(p)\equiv(2i+1)^{p-1}\pmod p$.
\paragraph{Statement recorded in the supplied source.}
Is the following equivalence valid for every integer $p\ge8$?
\[
 p\text{ is prime}\quad\Longleftrightarrow\quad
 \sum_{i=1}^{k(p)}r_i(p)=k(p).
\]
The displayed equality is an equality of integers, and each residue is chosen separately before summing. \sref{3361}{2}
\subsection{Short English statement}
Determine whether this finite sum of specified Fermat residues characterizes primes at least eight.
