\section{Elementary symmetric of a sum of matrices}\label{3027}
{\small\sffamily UnsolvedMath ID 3027 · Algebra}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For an $n\times n$ matrix $A$ over a field, define
\[
 S_k(A)=\sum_{\substack{I\subset\{1,\ldots,n\}\\|I|=k}}\det A[I,I],\qquad S_0(A)=1.
\]
Here $A[I,I]$ is the principal submatrix on rows and columns $I$; equivalently $\det(tI_n-A)=\sum_{k=0}^n(-1)^kS_k(A)t^{n-k}$. An integer partition of $k$ is a nonincreasing finite sequence of positive integers with sum $k$; the empty partition is allowed for $k=0$. Words in matrices are ordered products of the letters $A_1,\ldots,A_N$, so their order cannot in general be interchanged.
\paragraph{Statement recorded in the supplied source.}
Find a closed expression organized by integer partitions for
\[
 S_k(A_1+\cdots+A_N)
 \quad(0\leq N\leq k\leq n),
\]
in terms of matrix quantities built from the $A_i$, including elementary symmetric functions of their words as in the source's proposed framework. When $N=0$, the sum is the zero matrix. \sref{3027}{2}
\subsection{Short English statement}
Give a general partition formula for the elementary symmetric functions of a sum of matrices.
