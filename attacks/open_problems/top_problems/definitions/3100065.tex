\section{There is no n so that the only integer m with phi(n) = phi(m) is m=n}\label{3100065}
{\small\sffamily UnsolvedMath ID 3100065 · Number Theory}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For $n\in\mathbb N$, Euler's totient is
\[\varphi(n)=\#\{a\in\mathbb N:1\le a\le n,\ \gcd(a,n)=1\},\]
so $\varphi(1)=1$.

\paragraph{Statement recorded in the supplied source.}
For every $n\in\mathbb N$, does there exist $m\in\mathbb N\setminus\{n\}$ with $\varphi(m)=\varphi(n)$? Equivalently, can any value attained by the totient function have exactly one positive-integer preimage? \sref{3100065}{2}

\subsection{Short English statement}
Does every positive integer share its totient value with another positive integer?
