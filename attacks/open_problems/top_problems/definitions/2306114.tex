\section{Research Problems in Function Theory — Problem 6.114}\label{2306114}
{\small\sffamily UnsolvedMath ID 2306114 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. Write $\mathcal H^1$ for one-dimensional Hausdorff measure (length). A plane curve $\Gamma$ is regular in the stated sense if there is $C<\infty$ such that
\[\mathcal H^1(\Gamma\cap B(w,r))\leq Cr\qquad(w\in\mathbb C,\ r>0),\]
where $B(w,r)$ is an open disk. Let $f:\mathbb D\to f(\mathbb D)$ be a conformal map, that is, a holomorphic injective map; $f^{-1}(\Gamma)=\{z\in\mathbb D:f(z)\in\Gamma\}$.
\paragraph{Statement recorded in the supplied source.}
Must $\mathcal H^1(f^{-1}(\Gamma))<\infty$ hold for every regular curve $\Gamma$ and every such conformal map $f$? \sref{2306114}{2}
\subsection{Short English statement}
Does a conformal inverse image of every curve with uniformly linear local length have finite length inside the disk?
