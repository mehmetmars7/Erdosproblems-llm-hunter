\section{Research Problems in Function Theory — Problem 7.54}\label{2307054}
{\small\sffamily UnsolvedMath ID 2307054 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For $t\in\mathbb C$ put $\phi_t(z)=e^{tz}-1$, and define its iterates by $\phi_t^{\circ1}=\phi_t$ and $\phi_t^{\circ(n+1)}=\phi_t\circ\phi_t^{\circ n}$. The entire function of $t$ given by $F_n(t)=\phi_t^{\circ n}(-1)$ has Taylor expansion $F_n(t)=\sum_{k=0}^\infty a_{n,k}t^k$ at $0$. For example, $F_1(t)=e^{-t}-1$ and $F_2(t)=\exp(t(e^{-t}-1))-1$.
\paragraph{Statement recorded in the supplied source.}
Is $|a_{n,k}|\le1$ for every $n\ge1$ and every $k\ge0$? \sref{2307054}{2}
\subsection{Short English statement}
Are all Taylor coefficients, in the parameter, of every iterate evaluated at minus one bounded in modulus by one?
