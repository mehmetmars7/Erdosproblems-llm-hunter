\section{Research Problems in Function Theory — Problem 7.72}\label{2307072}
{\small\sffamily UnsolvedMath ID 2307072 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $U=\{z:|z|<1\}$ and $dA=dx\,dy$. Let
\[N=\{v\in L^\infty(U):\int_Uv\phi\,dA=0
\text{ for every }\phi\in\mathcal O(U)\text{ with }\int_U|\phi|\,dA<\infty\}.
\]
The pairing contains no complex conjugation. Extend $v$ by zero off $U$. Its Beurling transform, defined almost everywhere as a singular integral, is
\[(Bv)(z)=-\frac1\pi\lim_{\varepsilon\downarrow0}
\int_{U\setminus B(z,\varepsilon)}\frac{v(\zeta)}{(z-\zeta)^2}\,dA(\zeta).
\]
Let $P$ be all finite complex linear combinations of $z^j\bar z^k$, $j,k\ge0$. Norms for $v$ are essential suprema on $U$. For $v\in N$ the displayed transform vanishes off $\overline U$, by expanding its analytic kernel in $\zeta$, so the essential supremum of $Bv$ on the plane or on $U$ is the same.
\paragraph{Statement recorded in the supplied source.}
Is there a universal finite constant $C$ such that every $v\in N$ satisfies $Bv\in L^\infty$ and $\|Bv\|_\infty\le C\|v\|_\infty$? Separately, does this uniform assertion at least hold for every $v\in N\cap P$? \sref{2307072}{2}
\subsection{Short English statement}
Is the Beurling transform uniformly bounded on bounded functions annihilating all integrable holomorphic tests, or at least on the polynomial subclass?
