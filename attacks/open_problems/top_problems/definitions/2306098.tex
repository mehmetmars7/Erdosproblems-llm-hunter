\section{Research Problems in Function Theory — Problem 6.98}\label{2306098}
{\small\sffamily UnsolvedMath ID 2306098 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. For a nonconstant holomorphic function $f$ on a domain $D$, its valence is $v_D(f)=\sup_{w\in\mathbb C}n_D(w,f)\in\mathbb N\cup\{\infty\}$, where $n_D(w,f)$ counts the zeros of $f-w$ in $D$ with multiplicity. The function is $p$-valent if $v_D(f)\leq p$. Let $p,q\geq1$ be integers and let $f(z)=z^q+\sum_{n=q+1}^{\infty}a_nz^n$ be holomorphic and $p$-valent in $\mathbb D$. List its nonzero zeros, with multiplicity, as $\beta_1,\ldots,\beta_s$, where $0<|\beta_k|<1$. Put $t=p-q-s\geq0$ and define the coefficients $A_n$ by
\[F(z)=\frac{z^q}{(1-z)^{2q+2s}}\left(\frac{1+z}{1-z}\right)^{2t}
\prod_{k=1}^{s}\left(1+\frac{z}{|\beta_k|}\right)(1+|\beta_k|z)
=\sum_{n=q}^{\infty}A_nz^n.\]
An empty product is one. In particular $A_q=1$; the coefficients are defined by the displayed analytic product, including when $q>1$.
\paragraph{Statement recorded in the supplied source.}
Does $|a_n|\leq|A_n|$ hold for all integers $n>q$ and every such $f$? \sref{2306098}{2}
\subsection{Short English statement}
Does the indicated product, determined by the zeros and valence, majorize every coefficient of the function?
