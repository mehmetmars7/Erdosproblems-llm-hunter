\section{Research Problems in Function Theory — Problem 2.66}\label{2302066}
{\small\sffamily UnsolvedMath ID 2302066 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


For an entire function $f$, put $M(r,f)=\max_{|z|=r}|f(z)|$, $T(r,f)=(2\pi)^{-1}\int_0^{2\pi}\log^+|f(re^{i\theta})|\,d\theta$, and $\rho(f)=\limsup_{r\to\infty}\log\log M(r,f)/\log r$, where $\log^+t=\max(0,\log t)$. Let $\omega_1$ be the first uncountable ordinal. Work in ZFC, without assuming the continuum hypothesis.
\paragraph{Statement recorded in the supplied source.}
Does there exist a family of entire functions $(f_\alpha)_{\alpha<\omega_1}$ such that
\[
\lim_{r\to\infty}\frac{M(r,f_\alpha)}{M(r,f_\beta)}=0\quad(\alpha<\beta<\omega_1)
\]
and, for every entire function $f$, there is $\gamma<\omega_1$ for which
\[\lim_{r\to\infty}\frac{M(r,f)}{M(r,f_\gamma)}=0?\] \sref{2302066}{2}
\subsection{Short English statement}
Without the continuum hypothesis, can an increasing family of length the first uncountable ordinal eventually dominate the growth of every entire function?
