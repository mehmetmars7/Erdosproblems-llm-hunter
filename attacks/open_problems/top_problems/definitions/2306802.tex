\section{Research Problems in Function Theory — Problem 6.2$\prime$}\label{2306802}
{\small\sffamily UnsolvedMath ID 2306802 · Analysis}
\subsection{Definitions and mathematical statement}

\paragraph{Shared notation.}$\mathbb N=\{1,2,\ldots\}$, and $\mathbb Z,\mathbb Q,\mathbb R,\mathbb C$ denote the integers, rationals, reals and complex numbers. Any other conventions are specified locally.


Write $\mathbb D=\{z\in\mathbb C:|z|<1\}$ and $\mathbb T=\partial\mathbb D$. For $p>0$ and a holomorphic $f$ on $\mathbb D$, let $n_f(w)$ count the preimages of $w$ with multiplicity. The function is areally mean $p$-valent when
\[\int_0^{2\pi}\int_0^R n_f(\rho e^{i\theta})\rho\,d\rho\,d\theta\leq\pi pR^2\qquad(R>0).\]
Writing $f(z)=\sum_{j=0}^{\infty}a_jz^j$, put $\mu_p(f)=\max\{|a_j|:j\in\mathbb Z,\ 0\leq j\leq p\}$, and let $S(p)$ be the mean $p$-valent functions with $\mu_p(f)=1$.
For each $p>0$ and integer $n\geq1$, define $A_n^{(p)}=\sup_{f\in S(p)}|a_n|$.
\paragraph{Statement recorded in the supplied source.}
For each $p>0$, does a finite limit
\[\lim_{n\to\infty}\frac{A_n^{(p)}}{n^{2p-1}}=K_p\]
exist, and if it exists, what is its value? For comparison, the source reports existence of the corresponding limit for each fixed $f\in S(p)$ when $p>1/4$. \sref{2306802}{2}
\subsection{Short English statement}
Does the extremal coefficient magnitude in the normalized mean-valent class have a limiting power-law constant?
